class HistoricalStrategyReplay:
    """
    Replays historical candles through the existing strategy pipeline.

    Each assessment uses only candles whose closing times are at or before
    the assessment timestamp. No future candles are passed to the strategy.

    This component evaluates strategy signals. It does not simulate orders,
    entries, stop losses, take profits, or trade execution.
    """

    TIMEFRAME_MINUTES = {
        "H4": 240,
        "H1": 60,
        "M15": 15,
        "M5": 5,
    }

    def __init__(
        self,
        storage=None,
        strategy_orchestrator=None,
        minimum_candles=None,
    ):
        self.storage = (
            storage if storage is not None else HistoricalDataStorage()
        )
        self.strategy_orchestrator = (
            strategy_orchestrator
            if strategy_orchestrator is not None
            else StrategyOrchestrator()
        )

        defaults = {
            "H4": 20,
            "H1": 20,
            "M15": 20,
            "M5": 20,
        }

        if minimum_candles is not None:
            unknown = set(minimum_candles) - set(defaults)
            if unknown:
                raise ValueError(
                    f"Unsupported timeframe(s): {sorted(unknown)}"
                )

            for timeframe, count in minimum_candles.items():
                if not isinstance(count, int) or count < 1:
                    raise ValueError(
                        f"Minimum candle count for {timeframe} "
                        "must be a positive integer."
                    )
                defaults[timeframe] = count

        self.minimum_candles = defaults

    @staticmethod
    def _normalize_datetime(value, field_name):
        if value is None:
            return None

        timestamp = pd.Timestamp(value)

        if timestamp.tzinfo is None:
            raise ValueError(
                f"{field_name} must be timezone-aware."
            )

        return timestamp.tz_convert("UTC")

    @staticmethod
    def _validate_dataset(data, timeframe):
        if not isinstance(data, pd.DataFrame):
            raise TypeError(
                f"{timeframe} dataset must be a pandas DataFrame."
            )

        if data.empty:
            raise ValueError(
                f"{timeframe} historical dataset is empty."
            )

        required_columns = {
            "time", "open", "high", "low", "close"
        }

        missing = required_columns - set(data.columns)

        if missing:
            raise ValueError(
                f"{timeframe} dataset is missing columns: "
                f"{sorted(missing)}"
            )

        if data["time"].isna().any():
            raise ValueError(
                f"{timeframe} dataset contains missing timestamps."
            )

        timestamps = pd.to_datetime(data["time"], utc=True)

        if timestamps.duplicated().any():
            raise ValueError(
                f"{timeframe} dataset contains duplicate timestamps."
            )

        if not timestamps.is_monotonic_increasing:
            raise ValueError(
                f"{timeframe} dataset is not chronologically ordered."
            )

        result = data.copy()
        result["time"] = timestamps

        return result

    @staticmethod
    def _latest_contiguous_m5_segment(completed_m5):
        """
        Return the latest uninterrupted sequence of completed M5 candles.

        A new segment begins whenever adjacent candle opening timestamps
        are not exactly five minutes apart. This handles weekend closures
        and missing historical candles without weakening detector validation.
        """

        if completed_m5.empty:
            return completed_m5.copy()

        result = completed_m5.reset_index(drop=True).copy()

        expected_interval = pd.Timedelta(minutes=5)
        differences = result["time"].diff()

        gap_positions = [
            position
            for position, difference in enumerate(differences)
            if position > 0 and difference != expected_interval
        ]

        if gap_positions:
            latest_segment_start = gap_positions[-1]
            result = result.iloc[latest_segment_start:].copy()

        return result.reset_index(drop=True)

    def load_datasets(self, symbol):
        """Load and validate all four historical timeframe datasets."""

        if not isinstance(symbol, str) or not symbol.strip():
            raise ValueError("Symbol must be a non-empty string.")

        datasets = {}

        for timeframe in self.TIMEFRAME_MINUTES:
            data = self.storage.load_dataset(
                symbol.strip(), timeframe
            )
            datasets[timeframe] = self._validate_dataset(
                data, timeframe
            )

        return datasets

    @classmethod
    def _completed_candles(
        cls,
        data,
        timeframe,
        assessment_time,
    ):
        duration = pd.Timedelta(
            minutes=cls.TIMEFRAME_MINUTES[timeframe]
        )

        closing_times = data["time"] + duration

        completed_mask = closing_times <= assessment_time

        if "is_forming_candle" in data.columns:
            completed_mask &= ~data["is_forming_candle"].fillna(
                True
            ).astype(bool)

        return data.loc[completed_mask].copy()

    def _assess(
        self,
        datasets,
        assessment_time,
    ):
        synchronized_inputs = {}

        for timeframe, data in datasets.items():
            completed = self._completed_candles(
                data,
                timeframe,
                assessment_time,
            )

            if timeframe == "M5":
                completed = self._latest_contiguous_m5_segment(
                    completed
                )

            if len(completed) < self.minimum_candles[timeframe]:
                return None, "insufficient_history"

            latest_close = (
                completed["time"].iloc[-1]
                + pd.Timedelta(
                    minutes=self.TIMEFRAME_MINUTES[timeframe]
                )
            )

            maximum_age = pd.Timedelta(
                minutes=self.TIMEFRAME_MINUTES[timeframe]
            )

            if assessment_time - latest_close > maximum_age:
                return None, "stale_data"

            synchronized_inputs[timeframe] = completed

        pipeline_result = self.strategy_orchestrator.analyze(
            h4_candles=synchronized_inputs["H4"],
            h1_candles=synchronized_inputs["H1"],
            m15_candles=synchronized_inputs["M15"],
            m5_candles=synchronized_inputs["M5"],
            assessment_time=assessment_time,
        )

        assessment = ReplayAssessment(
            assessment_time=assessment_time,
            alignment=pipeline_result.alignment,
            qualified_setup_count=len(
                pipeline_result.qualified_setups
            ),
            confirmed_setup_count=len(
                pipeline_result.confirmed_setups
            ),
            waiting_setup_count=len(
                pipeline_result.waiting_setups
            ),
            expired_setup_count=len(
                pipeline_result.expired_setups
            ),
            rejected_setup_count=len(
                pipeline_result.rejected_setups
            ),
        )

        return assessment, None

    def run(
        self,
        symbol="EURUSD",
        start_time=None,
        end_time=None,
        assessment_interval=1,
        max_assessments=None,
    ):
        """
        Replay the strategy chronologically using completed M5 candles.

        assessment_interval=1 evaluates every completed M5 candle.
        Larger values reduce the number of assessments for quicker trials.
        """

        if (
            not isinstance(assessment_interval, int)
            or assessment_interval < 1
        ):
            raise ValueError(
                "assessment_interval must be a positive integer."
            )

        if max_assessments is not None:
            if (
                not isinstance(max_assessments, int)
                or max_assessments < 1
            ):
                raise ValueError(
                    "max_assessments must be a positive integer."
                )

        start = self._normalize_datetime(
            start_time, "start_time"
        )
        end = self._normalize_datetime(
            end_time, "end_time"
        )

        if start is not None and end is not None and start > end:
            raise ValueError(
                "start_time cannot be later than end_time."
            )

        datasets = self.load_datasets(symbol)
        m5_data = datasets["M5"]

        m5_closes = (
            m5_data["time"]
            + pd.Timedelta(minutes=self.TIMEFRAME_MINUTES["M5"])
        )

        candidate_times = m5_closes.drop_duplicates().sort_values()

        if start is not None:
            candidate_times = candidate_times[
                candidate_times >= start
            ]

        if end is not None:
            candidate_times = candidate_times[
                candidate_times <= end
            ]

        assessments = []
        skipped_insufficient = 0
        skipped_stale = 0
        processed_candidates = 0

        for assessment_time in candidate_times:
            processed_candidates += 1

            if (
                processed_candidates - 1
            ) % assessment_interval != 0:
                continue

            assessment_time = pd.Timestamp(
                assessment_time
            ).tz_convert("UTC")

            assessment, skip_reason = self._assess(
                datasets,
                assessment_time,
            )

            if skip_reason == "insufficient_history":
                skipped_insufficient += 1
                continue

            if skip_reason == "stale_data":
                skipped_stale += 1
                continue

            assessments.append(assessment)

            if (
                max_assessments is not None
                and len(assessments) >= max_assessments
            ):
                break

        actual_start = (
            assessments[0].assessment_time
            if assessments else None
        )
        actual_end = (
            assessments[-1].assessment_time
            if assessments else None
        )

        return HistoricalReplayResult(
            symbol=symbol.strip(),
            start_time=actual_start,
            end_time=actual_end,
            assessment_count=len(assessments),
            skipped_insufficient_history=skipped_insufficient,
            skipped_stale_data=skipped_stale,
            assessments=tuple(assessments),
        )

"""Validates and synchronises H4, H1, M15 and M5 candle datasets.

The synchronizer:
- Requires timezone-aware candle timestamps.
- Converts timestamps to UTC.
- Rejects duplicate or chronologically unordered candles.
- Excludes candles that have not closed by assessment_time.
- Checks the age of the latest completed candle.
- Returns independent copies of the validated candle datasets.

This component does not fetch market data or execute trades.
"""

from dataclasses import dataclass
from typing import Optional

import pandas as pd


@dataclass(frozen=True)
class SynchronizedTimeframeData:
    """Validated candle datasets for the four strategy timeframes."""

    assessment_time: pd.Timestamp
    h4_candles: pd.DataFrame
    h1_candles: pd.DataFrame
    m15_candles: pd.DataFrame
    m5_candles: pd.DataFrame
    latest_candle_close_times: dict


class TimeframeDataSynchronizer:
    """Validates and synchronises four timeframe candle datasets."""

    TIMEFRAME_DURATIONS = {
        "H4": pd.Timedelta(hours=4),
        "H1": pd.Timedelta(hours=1),
        "M15": pd.Timedelta(minutes=15),
        "M5": pd.Timedelta(minutes=5),
    }

    def __init__(self, max_age_by_timeframe: Optional[dict] = None):
        """Initialise timeframe freshness limits.

        By default, the maximum permitted age of the latest completed
        candle is one timeframe duration.

        Custom limits can be supplied as a dictionary, for example:

            {
                "H4": pd.Timedelta(hours=8),
                "H1": pd.Timedelta(hours=2),
                "M15": pd.Timedelta(minutes=30),
                "M5": pd.Timedelta(minutes=10),
            }
        """

        self.max_age_by_timeframe = {}

        supplied_limits = max_age_by_timeframe or {}

        unknown_timeframes = (
            set(supplied_limits) - set(self.TIMEFRAME_DURATIONS)
        )

        if unknown_timeframes:
            raise ValueError(
                "Unsupported timeframe freshness limits: "
                f"{sorted(unknown_timeframes)}"
            )

        for timeframe, duration in self.TIMEFRAME_DURATIONS.items():
            maximum_age = supplied_limits.get(
                timeframe,
                duration,
            )

            try:
                maximum_age = pd.Timedelta(maximum_age)
            except (TypeError, ValueError) as exc:
                raise ValueError(
                    f"Invalid maximum age for {timeframe}."
                ) from exc

            if pd.isna(maximum_age) or maximum_age <= pd.Timedelta(0):
                raise ValueError(
                    f"Maximum age for {timeframe} must be positive."
                )

            self.max_age_by_timeframe[timeframe] = maximum_age

    @staticmethod
    def _normalise_assessment_time(assessment_time) -> pd.Timestamp:
        """Validate assessment_time and convert it to UTC."""

        try:
            timestamp = pd.Timestamp(assessment_time)
        except (TypeError, ValueError) as exc:
            raise ValueError(
                "assessment_time must be a valid timestamp."
            ) from exc

        if pd.isna(timestamp):
            raise ValueError(
                "assessment_time cannot be NaT."
            )

        if timestamp.tzinfo is None or timestamp.utcoffset() is None:
            raise ValueError(
                "assessment_time must be timezone-aware."
            )

        return timestamp.tz_convert("UTC")

    @staticmethod
    def _normalise_candle_times(
        candles: pd.DataFrame,
        timeframe: str,
    ) -> pd.DataFrame:
        """Validate candle timestamps and return a normalised copy."""

        if not isinstance(candles, pd.DataFrame):
            raise TypeError(
                f"{timeframe} candles must be a pandas DataFrame."
            )

        if candles.empty:
            raise ValueError(
                f"{timeframe} candle data cannot be empty."
            )

        if "time" not in candles.columns:
            raise ValueError(
                f"{timeframe} candles must contain a 'time' column."
            )

        normalised = candles.copy()

        timestamps = []

        for index, value in normalised["time"].items():
            try:
                timestamp = pd.Timestamp(value)
            except (TypeError, ValueError) as exc:
                raise ValueError(
                    f"{timeframe} contains an invalid timestamp "
                    f"at index {index}."
                ) from exc

            if pd.isna(timestamp):
                raise ValueError(
                    f"{timeframe} contains a missing timestamp "
                    f"at index {index}."
                )

            if (
                timestamp.tzinfo is None
                or timestamp.utcoffset() is None
            ):
                raise ValueError(
                    f"{timeframe} contains a timezone-naive timestamp "
                    f"at index {index}."
                )

            timestamps.append(timestamp.tz_convert("UTC"))

        normalised["time"] = pd.Series(
            pd.DatetimeIndex(timestamps),
            index=normalised.index,
        )

        if normalised["time"].duplicated().any():
            raise ValueError(
                f"{timeframe} contains duplicate candle timestamps."
            )

        if not normalised["time"].is_monotonic_increasing:
            raise ValueError(
                f"{timeframe} candle timestamps must be "
                "in chronological order."
            )

        return normalised

    def _process_timeframe(
        self,
        candles: pd.DataFrame,
        timeframe: str,
        assessment_time: pd.Timestamp,
    ):
        """Filter completed candles and validate their freshness."""

        if timeframe not in self.TIMEFRAME_DURATIONS:
            raise ValueError(
                f"Unsupported timeframe: {timeframe}"
            )

        normalised = self._normalise_candle_times(
            candles=candles,
            timeframe=timeframe,
        )

        candle_duration = self.TIMEFRAME_DURATIONS[timeframe]

        # Candle timestamps represent opening times.
        # A candle is completed only when its closing time is
        # at or before the assessment time.
        candle_close_times = (
            normalised["time"] + candle_duration
        )

        completed_mask = candle_close_times <= assessment_time

        completed_candles = normalised.loc[
            completed_mask
        ].copy()

        if completed_candles.empty:
            raise ValueError(
                f"{timeframe} has no completed candles at "
                "the supplied assessment_time."
            )

        latest_candle_open = completed_candles["time"].iloc[-1]

        latest_candle_close = (
            latest_candle_open + candle_duration
        )

        candle_age = (
            assessment_time - latest_candle_close
        )

        maximum_age = self.max_age_by_timeframe[timeframe]

        if candle_age > maximum_age:
            raise ValueError(
                f"{timeframe} candle data is stale. "
                f"Latest candle closed at "
                f"{latest_candle_close.isoformat()}, "
                f"assessment time is {assessment_time.isoformat()}, "
                f"and maximum permitted age is {maximum_age}."
            )

        return completed_candles, latest_candle_close

    def synchronize(
        self,
        h4_candles: pd.DataFrame,
        h1_candles: pd.DataFrame,
        m15_candles: pd.DataFrame,
        m5_candles: pd.DataFrame,
        assessment_time,
    ) -> SynchronizedTimeframeData:
        """Validate and synchronise all four timeframe datasets."""

        normalised_assessment_time = (
            self._normalise_assessment_time(assessment_time)
        )

        candle_inputs = {
            "H4": h4_candles,
            "H1": h1_candles,
            "M15": m15_candles,
            "M5": m5_candles,
        }

        processed = {}
        latest_close_times = {}

        for timeframe, candles in candle_inputs.items():
            completed_candles, latest_close = (
                self._process_timeframe(
                    candles=candles,
                    timeframe=timeframe,
                    assessment_time=normalised_assessment_time,
                )
            )

            processed[timeframe] = completed_candles
            latest_close_times[timeframe] = latest_close

        return SynchronizedTimeframeData(
            assessment_time=normalised_assessment_time,
            h4_candles=processed["H4"],
            h1_candles=processed["H1"],
            m15_candles=processed["M15"],
            m5_candles=processed["M5"],
            latest_candle_close_times=latest_close_times,
        )

    confirmation_break_index: Optional[int]
    confirmation_event: Optional[StructureBreak]
    candles_elapsed: int
    reason: str

class M5ConfirmationDetector:
    """
    Confirms qualified M15 setups using existing M5 BOS events.

    The detector:
    - Accepts only QUALIFIED M15 setups.
    - Requires an M5 BOS in the same direction.
    - Ignores M5 BOS events before M15 qualification.
    - Uses a maximum window of 12 eligible M5 candles.
    - Selects the earliest matching BOS.
    - Does not calculate BOS independently.
    - Does not place or manage trades.

    All supplied M5 candles must be completed candles.
    """

    TIMEFRAME_DURATION = timedelta(minutes=5)

    def __init__(self, confirmation_window_candles: int = 12):
        if (
            not isinstance(confirmation_window_candles, int)
            or isinstance(confirmation_window_candles, bool)
            or confirmation_window_candles < 1
        ):
            raise ValueError(
                "confirmation_window_candles must be a positive integer."
            )

        self.confirmation_window_candles = confirmation_window_candles

    @staticmethod
    def _to_utc_datetime(value, field_name: str) -> datetime:
        timestamp = pd.Timestamp(value)

        if pd.isna(timestamp):
            raise ValueError(f"{field_name} cannot be NaT.")

        if timestamp.tzinfo is None:
            raise ValueError(
                f"{field_name} must be timezone-aware."
            )

        return timestamp.to_pydatetime().astimezone(timezone.utc)

    @classmethod
    def _setup_identity(
        cls,
        setup,
        qualification_time: Optional[datetime],
    ) -> str:
        break_time = cls._to_utc_datetime(
            setup.break_time,
            "M15 setup break_time",
        )

        identity_parts = (
            str(setup.direction),
            break_time.isoformat(),
            str(setup.break_index),
            str(float(setup.broken_level)),
            (
                qualification_time.isoformat()
                if qualification_time is not None
                else "NOT_QUALIFIED"
            ),
        )

        raw_identity = "|".join(identity_parts)
        return hashlib.sha256(
            raw_identity.encode("utf-8")
        ).hexdigest()[:20]

    @staticmethod
    def _validate_inputs(
        m15_setups: M15SetupResult,
        m5_candles: pd.DataFrame,
        m5_market_structure: MarketStructureResult,
    ) -> None:
        if not hasattr(m15_setups, "setups"):
            raise TypeError(
                "m15_setups must provide a setups collection."
            )

        if not isinstance(m5_candles, pd.DataFrame):
            raise TypeError("m5_candles must be a pandas DataFrame.")

        if "time" not in m5_candles.columns:
            raise ValueError(
                "m5_candles must contain a 'time' column."
            )

        if not hasattr(m5_market_structure, "bos_events"):
            raise TypeError(
                "m5_market_structure must provide bos_events."
            )

        timeframe = str(
            getattr(m5_market_structure, "timeframe", "")
        ).upper()

        if timeframe not in {"M5", "5M"} and not timeframe.endswith(".M5"):
            raise ValueError(
                "m5_market_structure must represent the M5 timeframe."
            )

        if len(m5_candles) != getattr(
            m5_market_structure, "candle_count", len(m5_candles)
        ):
            raise ValueError(
                "M5 candle count does not match market structure candle_count."
            )

        candle_times = [
            M5ConfirmationDetector._to_utc_datetime(
                value,
                f"M5 candle time at index {index}",
            )
            for index, value in enumerate(m5_candles["time"])
        ]

        if any(
            current <= previous
            for previous, current in zip(
                candle_times,
                candle_times[1:],
            )
        ):
            raise ValueError(
                "M5 candle timestamps must be strictly increasing."
            )

        if len(set(candle_times)) != len(candle_times):
            raise ValueError("Duplicate M5 candle timestamps detected.")

        for previous, current in zip(
            candle_times,
            candle_times[1:],
        ):
            if current - previous != timedelta(minutes=5):
                raise ValueError(
                    "M5 candle timestamps must be spaced exactly 5 minutes apart."
                )

    @staticmethod
    def _validate_bos_events(
        bos_events,
        candle_times: list[datetime],
    ) -> None:
        for event in bos_events:
            if event.event_type != "BOS":
                raise ValueError(
                    "m5_market_structure.bos_events must contain BOS events only."
                )

            if event.direction not in {"BULLISH", "BEARISH"}:
                raise ValueError(
                    "BOS direction must be BULLISH or BEARISH."
                )

            if (
                not isinstance(event.break_index, int)
                or isinstance(event.break_index, bool)
                or not 0 <= event.break_index < len(candle_times)
            ):
                raise ValueError(
                    "BOS break_index is outside the M5 candle range."
                )

            event_time = M5ConfirmationDetector._to_utc_datetime(
                event.break_time,
                "BOS break_time",
            )

            expected_time = candle_times[event.break_index]

            if event_time != expected_time:
                raise ValueError(
                    "BOS break_time must match the opening time of its indexed M5 candle."
                )

    @staticmethod
    def _rejected(
        setup_id: str,
        direction: str,
        reason: str,
        qualification_time: Optional[datetime] = None,
    ) -> M5Confirmation:
        return M5Confirmation(
            setup_id=setup_id,
            direction=direction,
            state="REJECTED",
            qualification_time=qualification_time,
            confirmation_time=None,
            confirmation_break_time=None,
            confirmation_break_index=None,
            confirmation_event=None,
            candles_elapsed=0,
            reason=reason,
        )

    def _evaluate_setup(
        self,
        setup,
        candle_times: list[datetime],
        bos_events,
    ) -> M5Confirmation:
        direction = str(setup.direction)

        if direction not in {"BULLISH", "BEARISH"}:
            raise ValueError(
                f"Unsupported M15 setup direction: {direction}"
            )

        state = str(setup.state)

        if state != "QUALIFIED":
            setup_id = self._setup_identity(setup, None)

            return self._rejected(
                setup_id=setup_id,
                direction=direction,
                reason=(
                    f"M15 setup state is {state}; "
                    "only QUALIFIED setups can receive M5 confirmation."
                ),
            )

        if setup.qualification_time is None:
            raise ValueError(
                "A QUALIFIED M15 setup must have qualification_time."
            )

        qualification_time = self._to_utc_datetime(
            setup.qualification_time,
            "M15 qualification_time",
        )

        setup_id = self._setup_identity(
            setup,
            qualification_time,
        )

        eligible_indices = [
            index
            for index, candle_time in enumerate(candle_times)
            if candle_time >= qualification_time
        ]

        window_indices = eligible_indices[
            :self.confirmation_window_candles
        ]

        matching_events = [
            event
            for event in bos_events
            if (
                event.direction == direction
                and event.break_index in window_indices
            )
        ]

        matching_events.sort(key=lambda event: event.break_index)

        if matching_events:
            event = matching_events[0]

            return M5Confirmation(
                setup_id=setup_id,
                direction=direction,
                state="CONFIRMED",
                qualification_time=qualification_time,
                confirmation_time=(
                    candle_times[event.break_index]
                    + self.TIMEFRAME_DURATION
                ),
                confirmation_break_time=self._to_utc_datetime(
                    event.break_time,
                    "BOS break_time",
                ),
                confirmation_break_index=event.break_index,
                confirmation_event=event,
                candles_elapsed=event.break_index - window_indices[0] + 1,
                reason=(
                    f"{direction} M5 BOS confirmed the qualified "
                    "M15 setup within the confirmation window."
                ),
            )

        candles_elapsed = len(window_indices)

        if candles_elapsed >= self.confirmation_window_candles:
            return M5Confirmation(
                setup_id=setup_id,
                direction=direction,
                state="EXPIRED",
                qualification_time=qualification_time,
                confirmation_time=None,
                confirmation_break_time=None,
                confirmation_break_index=None,
                confirmation_event=None,
                candles_elapsed=candles_elapsed,
                reason=(
                    "No matching M5 BOS occurred within "
                    f"{self.confirmation_window_candles} eligible candles."
                ),
            )

        return M5Confirmation(
            setup_id=setup_id,
            direction=direction,
            state="WAITING",
            qualification_time=qualification_time,
            confirmation_time=None,
            confirmation_break_time=None,
            confirmation_break_index=None,
            confirmation_event=None,
            candles_elapsed=candles_elapsed,
            reason=(
                "Waiting for a matching M5 BOS. "
                f"{candles_elapsed}/{self.confirmation_window_candles} "
                "eligible candles have been evaluated."
            ),
        )

    def detect(
        self,
        m15_setups: M15SetupResult,
        m5_candles: pd.DataFrame,
        m5_market_structure: MarketStructureResult,
    ) -> M5ConfirmationResult:
        self._validate_inputs(
            m15_setups,
            m5_candles,
            m5_market_structure,
        )

        candle_times = [
            self._to_utc_datetime(
                value,
                f"M5 candle time at index {index}",
            )
            for index, value in enumerate(m5_candles["time"])
        ]

        bos_events = tuple(m5_market_structure.bos_events)

        self._validate_bos_events(
            bos_events,
            candle_times,
        )

        confirmations = tuple(
            self._evaluate_setup(
                setup,
                candle_times,
                bos_events,
            )
            for setup in m15_setups.setups
        )

        return M5ConfirmationResult(
            confirmations=confirmations,
        )

class M15SetupDetector:
    """
    Detect M15 retest and rejection setups from confirmed BOS events.

    Consumes MarketStructureResult produced by MarketStructureEngine.
    Does not independently detect swings or BOS events.

    This component does not place, modify, or manage trading orders.

    def __init__(
        self,
        atr_period: int = 14,
        retest_tolerance_atr: float = 0.10,
        expiry_candles: int = 20,
        alignment_engine: Optional[AlignmentEngine] = None,

        if isinstance(atr_period, bool) or not isinstance(
            atr_period, int
        ):
            raise TypeError("atr_period must be an integer.")

        if atr_period < 1:
            raise ValueError("atr_period must be at least 1.")

        if (
            isinstance(retest_tolerance_atr, bool)
            or not isinstance(retest_tolerance_atr, (int, float))
        ):
            raise TypeError(
                "retest_tolerance_atr must be numeric."
            )

        if (
            not math.isfinite(float(retest_tolerance_atr))
            or retest_tolerance_atr <= 0
        ):
            raise ValueError(
                "retest_tolerance_atr must be positive and finite."
            )

        if isinstance(expiry_candles, bool) or not isinstance(
            expiry_candles, int
        ):
            raise TypeError("expiry_candles must be an integer.")

        if expiry_candles < 1:
            raise ValueError(
                "expiry_candles must be at least 1."
            )

        self.atr_period = atr_period
        self.retest_tolerance_atr = float(
            retest_tolerance_atr
        )
        self.expiry_candles = expiry_candles

        self.atr_calculator = WilderATR(period=atr_period)
        self.alignment_engine = (
            alignment_engine or AlignmentEngine()

    @staticmethod
    def _to_utc_datetime(value, field_name: str) -> datetime:
        """Convert an aware timestamp to a UTC Python datetime."""

        try:
            timestamp = pd.Timestamp(value)
        except (TypeError, ValueError, OverflowError) as exc:
            raise ValueError(
                f"{field_name} must be a valid timezone-aware timestamp."
            ) from exc

        if pd.isna(timestamp) or timestamp.tzinfo is None:
            raise ValueError(
                f"{field_name} must be timezone-aware."
            )

        return timestamp.tz_convert("UTC").to_pydatetime()

    @staticmethod
    def _validate_inputs(
        candles: pd.DataFrame,
        market_structure: MarketStructureResult,
    ) -> None:

        if not isinstance(candles, pd.DataFrame):
            raise TypeError(
                "candles must be a pandas DataFrame."
            )

        if not isinstance(
            market_structure,
            MarketStructureResult,
        ):
            raise TypeError(
                "market_structure must be a MarketStructureResult."
            )

        if market_structure.timeframe != "M15":
            raise ValueError(
                "market_structure must contain M15 analysis."
            )

        if market_structure.candle_count != len(candles):
            raise ValueError(
                "Candle count does not match market-structure analysis."
            )

        if "time" not in candles.columns:
            raise ValueError(
                "candles must contain a time column."
            )

    @classmethod
    def _validate_event(
        cls,
        event,
        candles: pd.DataFrame,
    ) -> datetime:
        """Validate one BOS event and return its normalized break time."""

        if event.event_type != "BOS":
            raise ValueError(
                "Only BOS events can create M15 retest setups."
            )

        if event.direction not in ("BULLISH", "BEARISH"):
            raise ValueError(
                "Unsupported BOS direction."
            )

        if (
            isinstance(event.break_index, bool)
            or not isinstance(event.break_index, int)
        ):
            raise TypeError(
                "BOS break_index must be an integer."
            )

        if not 0 <= event.break_index < len(candles):
            raise ValueError(
                "BOS break_index is outside candle data."
            )

        if (
            isinstance(event.level_price, bool)
            or not isinstance(event.level_price, (int, float))
            or not math.isfinite(float(event.level_price))
            or event.level_price <= 0
        ):
            raise ValueError(
                "BOS level_price must be positive and finite."
            )

        break_time = cls._to_utc_datetime(
            event.break_time,
            "BOS break_time",
        )

        candle_time = cls._to_utc_datetime(
            candles.iloc[event.break_index]["time"],
            "BOS candle time",
        )

        if break_time != candle_time:
            raise ValueError(
                "BOS break_time does not match its candle index."
            )

        return break_time

    def _alignment_permitted(
        self,
        h4_snapshots: tuple,
        h1_snapshots: tuple,
        assessment_time: datetime,
        direction: str,
    ) -> tuple[bool, str]:

        alignment = self.alignment_engine.evaluate(
            h4_snapshots=h4_snapshots,
            h1_snapshots=h1_snapshots,
            assessment_time=assessment_time,
        )

        if (
            alignment.h4_known_at_time is not None
            and alignment.h4_known_at_time > assessment_time
        ):
            raise RuntimeError(
                "H4 alignment lookahead detected."
            )

        if (
            alignment.h1_known_at_time is not None
            and alignment.h1_known_at_time > assessment_time
        ):
            raise RuntimeError(
                "H1 alignment lookahead detected."
            )

        permitted = (
            alignment.directional_permission
            and alignment.permitted_direction == direction
        )

        return permitted, alignment.reason

    @staticmethod
    def _build_setup(
        *,
        event,
        break_time: datetime,
        decision_time: datetime,
        state: SetupState,
        reason: str,
        atr_value: Optional[float] = None,
        tolerance: Optional[float] = None,
        retest_time: Optional[datetime] = None,
        qualification_time: Optional[datetime] = None,
        expiry_time: Optional[datetime] = None,
        candles_elapsed: int = 0,

        return M15Setup(
            direction=event.direction,
            state=state,
            break_time=break_time,
            decision_time=decision_time,
            break_index=event.break_index,
            broken_level=float(event.level_price),
            atr_value=atr_value,
            retest_tolerance=tolerance,
            retest_time=retest_time,
            qualification_time=qualification_time,
            expiry_time=expiry_time,
            candles_elapsed=candles_elapsed,
            reason=reason,
        )

    def _process_event(
        self,
        event,
        candles: pd.DataFrame,
        atr_values: pd.Series,
        h4_snapshots: tuple,
        h1_snapshots: tuple,
    ) -> M15Setup:

        break_time = self._validate_event(event, candles)

        decision_time = (
            break_time + self.TIMEFRAME_DURATION
        )

        permitted, alignment_reason = (
            self._alignment_permitted(
                h4_snapshots,
                h1_snapshots,
                decision_time,
                event.direction,
            )
        )

        if not permitted:
            return self._build_setup(
                event=event,
                break_time=break_time,
                decision_time=decision_time,
                state="REJECTED",
                reason=(
                    "BOS rejected by H4/H1 alignment: "
                    f"{alignment_reason}"
                ),
            )

        atr_value = atr_values.iloc[event.break_index]

        if (
            pd.isna(atr_value)
            or not math.isfinite(float(atr_value))
            or float(atr_value) <= 0
        ):
            return self._build_setup(
                event=event,
                break_time=break_time,
                decision_time=decision_time,
                state="REJECTED",
                reason=(
                    "ATR was unavailable or invalid at BOS confirmation."
                ),
            )

        atr_value = float(atr_value)

        # Tolerance is frozen using ATR at BOS confirmation.
        tolerance = (
            atr_value * self.retest_tolerance_atr
        )

        level = float(event.level_price)

        lower_boundary = level - tolerance
        upper_boundary = level + tolerance

        retest_time = None

        for index in range(
            event.break_index + 1,
            len(candles),
        ):

            elapsed = index - event.break_index
            candle = candles.iloc[index]

            candle_time = candle["time"].to_pydatetime()

            candle_close_time = (
                candle_time + self.TIMEFRAME_DURATION
            )

            # Reassess directional permission at every completed
            # M15 candle close. The alignment engine filters out
            # snapshots that were not yet knowable at that time.
            still_permitted, current_reason = (
                self._alignment_permitted(
                    h4_snapshots,
                    h1_snapshots,
                    candle_close_time,
                    event.direction,
                )
            )

            if not still_permitted:
                return self._build_setup(
                    event=event,
                    break_time=break_time,
                    decision_time=decision_time,
                    state="INVALIDATED",
                    reason=(
                        "H4/H1 directional permission was lost or "
                        f"reversed: {current_reason}"
                    ),
                    atr_value=atr_value,
                    tolerance=tolerance,
                    retest_time=retest_time,
                    candles_elapsed=elapsed,
                )

            close = float(candle["close"])
            high = float(candle["high"])
            low = float(candle["low"])

            # Invalidation takes priority over retest/rejection.
            if event.direction == "BULLISH":
                invalidated = close < lower_boundary
            else:
                invalidated = close > upper_boundary

            if invalidated:
                return self._build_setup(
                    event=event,
                    break_time=break_time,
                    decision_time=decision_time,
                    state="INVALIDATED",
                    reason=(
                        "Completed M15 candle closed beyond the "
                        "invalidation boundary."
                    ),
                    atr_value=atr_value,
                    tolerance=tolerance,
                    retest_time=retest_time,
                    candles_elapsed=elapsed,
                )

            if retest_time is None:

                previous_close = float(
                    candles.iloc[index - 1]["close"]
                )

                intersects_zone = (
                    low <= upper_boundary
                    and high >= lower_boundary
                )

                approached_from_correct_side = (
                    previous_close > level
                    if event.direction == "BULLISH"
                    else previous_close < level
                )

                if (
                    intersects_zone
                    and approached_from_correct_side
                ):
                    # Record the completed retest candle.
                    # It cannot also confirm rejection.
                    retest_time = candle_close_time

            else:

                rejection_confirmed = (
                    close > level
                    if event.direction == "BULLISH"
                    else close < level
                )

                if rejection_confirmed:
                    return self._build_setup(
                        event=event,
                        break_time=break_time,
                        decision_time=decision_time,
                        state="QUALIFIED",
                        reason=(
                            "Retest detected and a subsequent completed "
                            "M15 candle confirmed rejection."
                        ),
                        atr_value=atr_value,
                        tolerance=tolerance,
                        retest_time=retest_time,
                        qualification_time=candle_close_time,
                        candles_elapsed=elapsed,
                    )

            # Expiry is checked AFTER processing the candle.
            # This allows the 20th candle to qualify, but prevents
            # a retest first detected on candle 20 from remaining active.
            if elapsed == self.expiry_candles:
                return self._build_setup(
                    event=event,
                    break_time=break_time,
                    decision_time=decision_time,
                    state="EXPIRED",
                    reason=(
                        "Setup reached its configured expiry without "
                        "confirmed rejection."
                    ),
                    atr_value=atr_value,
                    tolerance=tolerance,
                    retest_time=retest_time,
                    expiry_time=candle_close_time,
                    candles_elapsed=elapsed,
                )

        if retest_time is not None:
            state: SetupState = "RETEST_DETECTED"
            reason = (
                "Retest detected. A subsequent rejection candle "
                "has not yet been confirmed."
            )
        else:
            state = "WAITING_FOR_RETEST"
            reason = (
                "Waiting for price to return to the retest zone."
            )

        candles_elapsed = min(
            len(candles) - 1 - event.break_index,
            self.expiry_candles,
        )

        return self._build_setup(
            event=event,
            break_time=break_time,
            decision_time=decision_time,
            state=state,
            reason=reason,
            atr_value=atr_value,
            tolerance=tolerance,
            retest_time=retest_time,
            candles_elapsed=candles_elapsed,
        )

    def detect(
        self,
        candles: pd.DataFrame,
        market_structure: MarketStructureResult,
        h4_snapshots,
        h1_snapshots,
    ) -> M15SetupResult:
        """
        Evaluate all confirmed M15 BOS events.

        Candles must be completed M15 candles.
        H4/H1 snapshots are evaluated using their known-at timestamps.

        No trading orders are placed or managed.
        """

        self._validate_inputs(
            candles,
            market_structure,
        )

        # Public calculation performs the ATR calculator's full
        # candle validation. Do not call its private validator.
        atr_values = self.atr_calculator.calculate(candles)

        # Normalize a separate working copy for detector operations.
        working_candles = candles.copy(deep=True)

        working_candles["time"] = pd.to_datetime(
            working_candles["time"],
            utc=True,
            errors="coerce",
        )

        for column in ("high", "low", "close"):
            working_candles[column] = pd.to_numeric(
                working_candles[column],
                errors="coerce",
            )

        # Materialize iterables once. This prevents generators from
        # being exhausted when multiple BOS events are evaluated.
        try:
            h4_snapshot_list = tuple(h4_snapshots)
            h1_snapshot_list = tuple(h1_snapshots)
        except TypeError as exc:
            raise TypeError(
                "H4 and H1 snapshots must be iterable."
            ) from exc

        ordered_events = sorted(
            market_structure.bos_events,
            key=lambda event: (
                event.break_index,
                event.direction,
            ),
        )

        setups = []

        for event in ordered_events:
            setups.append(
                self._process_event(
                    event=event,
                    candles=working_candles,
                    atr_values=atr_values,
                    h4_snapshots=h4_snapshot_list,
                    h1_snapshots=h1_snapshot_list,
                )
            )

        return M15SetupResult(
            setups=tuple(setups)
        )

class MarketStructureEngine:
    """
    Coordinates the existing market-structure components.

    Processing stages:

    1. Fractal swing detection.
    2. Swing classification.
    3. Trend classification.
    4. BOS detection.
    5. Protected swing tracking.
    6. Protected level registration.
    7. CHoCH detection.
    8. Chronological protected-level retirement.

    Detection remains separate from registry management.

    No trading orders are created or executed.
    """

    def __init__(
        self,
        left_bars: int = 2,
        right_bars: int = 2,
        price_tolerance: float = 0.0,
    ) -> None:

        self.fractal_detector = FractalSwingDetector(
            left_bars=left_bars,
            right_bars=right_bars,
        )

        self.structure_classifier = StructureClassifier(
            price_tolerance=price_tolerance,
        )

        self.trend_classifier = TrendClassifier()
        self.bos_detector = BOSDetector()
        self.protected_swing_tracker = ProtectedSwingTracker()
        self.choch_detector = CHoCHDetector()
        self.alignment_engine = AlignmentEngine()

    @staticmethod
    def _protected_identity(protected: ProtectedSwing) -> tuple:
        swing = protected.swing.swing

        return (
            protected.trend,
            swing.swing_type,
            swing.swing_index,
            swing.confirmation_index,
        )

    @staticmethod
    def _level_identity(level: ProtectedLevel) -> tuple:
        swing = level.swing.swing

        return (
            level.trend,
            swing.swing_type,
            swing.swing_index,
            swing.confirmation_index,
        )

    @staticmethod
    def _event_identity(event: StructureBreak) -> tuple:
        swing = event.level_swing.swing

        return (
            event.prior_trend,
            swing.swing_type,
            swing.swing_index,
            swing.confirmation_index,
        )

    @staticmethod
    def _protected_sort_key(protected: ProtectedSwing) -> tuple:
        return (
            protected.established_by.break_index,
            protected.swing.swing.confirmation_index,
            protected.swing.swing.swing_index,
            protected.trend,
        )

    @staticmethod
    def _event_sort_key(event: StructureBreak) -> tuple:
        return (
            event.break_index,
            event.break_time,
            event.direction,
        )

    def _detect_choch_events(
        self,
        candles: pd.DataFrame,
        protected_swings: tuple[ProtectedSwing, ...],
    ) -> tuple[StructureBreak, ...]:
        """
        Detect CHoCH events against all structurally established levels.

        This is a detection pass only. It does not mutate the final
        chronological registry.

        CHoCHDetector checks each level's activation_index, so a
        protected level cannot trigger before its BOS activation.
        """

        detection_registry = ProtectedLevelRegistry()

        ordered_protected = sorted(
            protected_swings,
            key=self._protected_sort_key,
        )

        for protected in ordered_protected:
            detection_registry.register([protected])

        events = self.choch_detector.detect(
            candles=candles,
            registry=detection_registry,
        )

        return tuple(
            sorted(events, key=self._event_sort_key)
        )

    def _replay_registry(
        self,
        protected_swings: tuple[ProtectedSwing, ...],
        choch_events: tuple[StructureBreak, ...],
    ) -> ProtectedLevelRegistry:
        """
        Reconstruct registry state in chronological candle order.

        BOS events register protected levels on their break candles.
        Their activation_index determines when they become usable.

        CHoCH events retire matching active levels on their own
        confirmation candles.

        BOS registration is processed before CHoCH retirement when
        both events occur on the same candle. A newly registered
        level cannot trigger on that candle because activation occurs
        on the following candle.
        """

        registry = ProtectedLevelRegistry()

        protected_by_index = defaultdict(list)
        choch_by_index = defaultdict(list)

        for protected in protected_swings:
            break_index = protected.established_by.break_index

            protected_by_index[break_index].append(protected)

        for event in choch_events:
            choch_by_index[event.break_index].append(event)

        timeline = sorted(
            set(protected_by_index) | set(choch_by_index)
        )

        for candle_index in timeline:

            current_protected = sorted(
                protected_by_index.get(candle_index, []),
                key=self._protected_sort_key,
            )

            # Register BOS-established protected levels first.
            for protected in current_protected:
                registry.register([protected])

            current_choch_events = sorted(
                choch_by_index.get(candle_index, []),
                key=self._event_sort_key,
            )

            # Retire the exact level broken by each CHoCH event.
            for event in current_choch_events:

                identity = self._event_identity(event)

                matching_levels = [
                    level
                    for level in registry.get_active_levels(
                        as_of_index=candle_index
                    )
                    if self._level_identity(level) == identity
                ]

                if len(matching_levels) != 1:
                    raise RuntimeError(
                        "CHoCH event could not be matched to exactly "
                        "one active protected level. "
                        f"Identity: {identity}; "
                        f"candle index: {candle_index}."
                    )

                registry.retire(
                    matching_levels[0],
                    event,
                )

        return registry

    def analyze(
        self,
        candles: pd.DataFrame,
        timeframe: str = "H1",
    ) -> MarketStructureResult:
        """
        Perform complete market-structure analysis for one timeframe.

        Supported timeframes are defined by BOSDetector.
        Candles must contain completed historical OHLC data.
        """

        if not isinstance(candles, pd.DataFrame):
            raise TypeError(
                "candles must be a pandas DataFrame."
            )

        if candles.empty:
            raise ValueError(
                "Candle dataframe cannot be empty."
            )

        if len(candles) < 2:
            raise ValueError(
                "At least two candles are required for "
                "market-structure analysis."
            )

        if not isinstance(timeframe, str):
            raise TypeError(
                "timeframe must be a string."
            )

        if timeframe not in BOSDetector.TIMEFRAME_DURATIONS:
            raise ValueError(
                f"Unsupported timeframe: {timeframe}."
            )

        # Stage 1: Detect confirmed fractal swings.
        swings = tuple(
            self.fractal_detector.detect(candles)
        )

        # Stage 2: Classify confirmed swings.
        classified_swings = tuple(
            self.structure_classifier.classify(swings)
        )

        # Stage 3: Generate chronological trend snapshots.
        trend_snapshots = tuple(
            self.trend_classifier.classify(classified_swings)
        )

        # Stage 4: Detect confirmed structural breaks.
        bos_events = tuple(
            sorted(
                self.bos_detector.detect(
                    candles=candles,
                    swings=classified_swings,
                    trend_snapshots=trend_snapshots,
                    timeframe=timeframe,
                ),
                key=self._event_sort_key,
            )
        )

        # Stage 5: Identify protected swings established by BOS.
        protected_swings = tuple(
            sorted(
                self.protected_swing_tracker.track(
                    swings=classified_swings,
                    bos_events=bos_events,
                ),
                key=self._protected_sort_key,
            )
        )

        # Stage 6 and 7: Detect CHoCH against established levels.
        choch_events = self._detect_choch_events(
            candles=candles,
            protected_swings=protected_swings,
        )

        # Stage 8: Reconstruct the final registry chronologically.
        registry = self._replay_registry(
            protected_swings=protected_swings,
            choch_events=choch_events,
        )

        return MarketStructureResult(
            timeframe=timeframe,
            candle_count=len(candles),

            swings=swings,
            classified_swings=classified_swings,
            trend_snapshots=trend_snapshots,

            bos_events=bos_events,
            protected_swings=protected_swings,
            choch_events=choch_events,

            protected_levels=registry.levels,
            active_protected_levels=tuple(
                registry.get_active_levels()
            ),
            retired_protected_levels=tuple(
                registry.get_retired_levels()
            ),
        )

    def analyze_multi_timeframe(
        self,
        h4_candles: pd.DataFrame,
        h1_candles: pd.DataFrame,
        assessment_time,
    ) -> MultiTimeframeMarketStructureResult:
        """
        Analyze H4 and H1 independently, then evaluate alignment.

        Directional permission is granted only when both timeframes
        have eligible, confirmed and matching directional trends.
        """

        h4_analysis = self.analyze(
            candles=h4_candles,
            timeframe="H4",
        )

        h1_analysis = self.analyze(
            candles=h1_candles,
            timeframe="H1",
        )

        alignment = self.alignment_engine.evaluate(
            h4_snapshots=h4_analysis.trend_snapshots,
            h1_snapshots=h1_analysis.trend_snapshots,
            assessment_time=assessment_time,
        )

        return MultiTimeframeMarketStructureResult(
            h4_analysis=h4_analysis,
            h1_analysis=h1_analysis,
            alignment=alignment,
        )

        latest_close - duration * (count - index)

        h4_candles,
        h1_candles,
        m15_candles,
        m5_candles,
        assessment_time,
    ):
        self.calls.append(
            {
                "h4_candles": h4_candles,
                "h1_candles": h1_candles,
                "m15_candles": m15_candles,
                "m5_candles": m5_candles,
                "assessment_time": assessment_time,
            }
        )

        datasets = {
            "H4": h4_candles,
            "H1": h1_candles,
            "M15": m15_candles,
            "M5": m5_candles,
        }

        for timeframe, candles in datasets.items():
            if not isinstance(candles, pd.DataFrame):
                raise TypeError(
                    f"{timeframe} candles must be a pandas DataFrame."
                )

            if candles.empty:
                raise ValueError(
                    f"{timeframe} candles cannot be empty."
                )

        if self.synchronized_data is None:
            self.synchronized_data = SimpleNamespace(
                assessment_time=assessment_time,
                h4_candles=h4_candles.copy(),
                h1_candles=h1_candles.copy(),
                m15_candles=m15_candles.copy(),
                m5_candles=m5_candles.copy(),
                latest_candle_close_times={},
            )

        return self.synchronized_data

from datetime import datetime, timedelta, timezone
from types import SimpleNamespace

import pandas as pd
import pytest

from research.swing_detection.break_models import StructureBreak
from strategy.m5_confirmation_detector import (
    M5ConfirmationDetector,
)


UTC = timezone.utc
BASE_TIME = datetime(2026, 1, 1, 10, 0, tzinfo=UTC)


def make_candles(count, start_time=BASE_TIME):
    return pd.DataFrame(
        {
            "time": [
                start_time + timedelta(minutes=5 * index)
                for index in range(count)
            ],
            "open": [100.0] * count,
            "high": [101.0] * count,
            "low": [99.0] * count,
            "close": [100.5] * count,
        }
    )


def make_setup(
    direction="BULLISH",
    state="QUALIFIED",
    qualification_time=BASE_TIME,
    break_time=None,
    break_index=1,
    broken_level=100.0,
):
    if break_time is None:
        break_time = BASE_TIME - timedelta(minutes=15)

    return SimpleNamespace(
        direction=direction,
        state=state,
        break_time=break_time,
        break_index=break_index,
        broken_level=broken_level,
        qualification_time=qualification_time,
    )


def make_bos(
    direction="BULLISH",
    break_index=0,
    break_time=None,
    event_type="BOS",
):
    if break_time is None:
        break_time = (
            BASE_TIME + timedelta(minutes=5 * break_index)
        )

    return StructureBreak(
        event_type=event_type,
        direction=direction,
        break_time=break_time,
        break_index=break_index,
        level_price=100.0,
        level_swing=None,
        close_price=101.0,
        prior_trend="BULLISH",
    )


def make_market_structure(
    candle_count,
    bos_events=None,
    timeframe="M5",
):
    return SimpleNamespace(
        timeframe=timeframe,
        candle_count=candle_count,
        bos_events=tuple(bos_events or []),
    )


def run_detector(
    setup,
    candles,
    bos_events=None,
    window=12,
):
    detector = M5ConfirmationDetector(
        confirmation_window_candles=window,
    )

    m15_result = SimpleNamespace(setups=(setup,))

    market_structure = make_market_structure(
        candle_count=len(candles),
        bos_events=bos_events,
    )

    return detector.detect(
        m15_setups=m15_result,
        m5_candles=candles,
        m5_market_structure=market_structure,
    )
